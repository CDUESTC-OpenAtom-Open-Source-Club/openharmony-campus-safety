cmake_minimum_required(VERSION 3.5.0)
project(xiaoan)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../" ABSOLUTE)
set(MNN_SOURCE "${REPO_ROOT}/.regression/vendor/MNN")
if(NOT EXISTS "${MNN_SOURCE}/transformers/llm/engine/include/llm/llm.hpp")
  message(FATAL_ERROR "Run scripts/build-xiaoan-runtime.ps1 before building the app")
endif()
add_library(MNN SHARED IMPORTED)
set_target_properties(MNN PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/../../../libs/${OHOS_ARCH}/libMNN.so")
add_library(xiaoan SHARED xiaoan.cpp)
target_include_directories(xiaoan PRIVATE "${MNN_SOURCE}/include" "${MNN_SOURCE}/transformers/llm/engine/include")
target_link_libraries(xiaoan PUBLIC MNN libace_napi.z.so)
